import React, { useMemo } from 'react';

interface SubtitleStyle {
  fontFamily?: string;
  fontSize: number;
  color: string;
  outlineColor?: string;
  outlineWidth?: number;
  shadowColor?: string;
  shadowWidth?: number;
  bold?: boolean;
  italic?: boolean;
  alignment?: number;
  marginL?: number;
  marginR?: number;
  marginV?: number;
}

interface Props {
  englishText: string;
  chineseText: string;
  englishStyle: SubtitleStyle;
  chineseStyle: SubtitleStyle;
  visible?: boolean;
}

const CSSSubtitleRenderer: React.FC<Props> = ({
  englishText,
  chineseText,
  englishStyle,
  chineseStyle,
  visible = true,
}) => {
  const enStyle = useMemo(() => buildTextStyle(englishStyle, false), [englishStyle]);
  const zhStyle = useMemo(() => buildTextStyle(chineseStyle, true), [chineseStyle]);
  const enPos = useMemo(() => buildPositionStyle(englishStyle), [englishStyle]);
  const zhPos = useMemo(() => buildPositionStyle(chineseStyle), [chineseStyle]);

  if (!visible || (!englishText && !chineseText)) return null;

  return (
    <div className="subtitle-overlay" style={{ position: 'absolute', inset: 0, pointerEvents: 'none', zIndex: 10 }}>
      {chineseText && (
        <div className="subtitle-line zh" style={{ ...zhStyle, ...zhPos }}>
          {chineseText}
        </div>
      )}
      {englishText && (
        <div className="subtitle-line en" style={{ ...enStyle, ...enPos }}>
          {englishText}
        </div>
      )}
    </div>
  );
};

// 位置样式
function buildPositionStyle(style: SubtitleStyle) {
  const align = style.alignment || 2;
  const vertical = style.marginV || 40;
  const marginL = style.marginL || 60;
  const marginR = style.marginR || 60;

  const pos: React.CSSProperties = {};

  // 垂直位置
  if (align >= 7) pos.top = `${vertical}px`;
  else if (align >= 4) {
    pos.top = '50%';
    pos.transform = 'translateY(-50%)';
  } else {
    pos.bottom = `${vertical}px`;
  }

  // 水平位置
  const hAlign = (align - 1) % 3;
  if (hAlign === 1) {
    pos.left = '50%';
    pos.transform = (pos.transform || '') + ' translateX(-50%)';
  } else if (hAlign === 0) {
    pos.left = `${marginL}px`;
  } else {
    pos.right = `${marginR}px`;
  }

  pos.transform = (pos.transform || '').trim() || undefined;
  return pos;
}

// 文字样式
function buildTextStyle(style: SubtitleStyle, isChinese: boolean) {
  const outlineW = style.outlineWidth || 2;
  const outlineC = style.outlineColor || '#000000';
  const shadowW = style.shadowWidth || 0;
  const shadowC = style.shadowColor || '#000000';

  const shadows = [
    `${outlineW}px 0 0 ${outlineC}`,
    `${-outlineW}px 0 0 ${outlineC}`,
    `0 ${outlineW}px 0 ${outlineC}`,
    `0 ${-outlineW}px 0 ${outlineC}`,
    `${outlineW}px ${outlineW}px 0 ${outlineC}`,
    `${-outlineW}px ${-outlineW}px 0 ${outlineC}`,
  ];
  if (shadowW > 0) shadows.push(`${shadowW}px ${shadowW}px ${shadowW}px ${shadowC}`);

  return {
    fontFamily: isChinese 
      ? `"${style.fontFamily || 'Microsoft YaHei'}", "SimHei", "SimSun", sans-serif`
      : `"${style.fontFamily || 'Arial'}", sans-serif`,
    fontSize: `${style.fontSize || 48}px`,
    color: style.color || '#FFFFFF',
    fontWeight: style.bold ? 700 : 400,
    fontStyle: style.italic ? 'italic' : 'normal',
    textShadow: shadows.join(', '),
    textAlign: 'center' as const,
    lineHeight: '1.3',
    userSelect: 'none' as const,
  } as React.CSSProperties;
}

export default CSSSubtitleRenderer;
